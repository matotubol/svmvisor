19.2.2 ASL Name and Pathname Terms
// Name and path characters supported
LeadNameChar :=
‘A’-‘Z’ | ‘a’-‘z’ | ‘_’
DigitChar :=
‘0’-‘9’
NameChar :=
DigitChar | LeadNameChar
RootChar :=
‘'
ParentPrefixChar :=
‘^’
PathSeparatorChar :=
‘.’
CommaChar :=
‘,’
