20.2.3 Data Objects Encoding
ComputationalData := ByteConst | WordConst | DWordConst | QWordConst | String | ConstObj | Revi-
sionOp | DefBuffer
DataObject := ComputationalData | DefPackage | DefVarPackage
DataRefObject := DataObject | ObjectReference
ByteConst := BytePrefix ByteData
BytePrefix := 0x0A
WordConst := WordPrefix WordData
WordPrefix := 0x0B
DWordConst := DWordPrefix DWordData
DWordPrefix := 0x0C
QWordConst := QWordPrefix QWordData
QWordPrefix := 0x0E
String := StringPrefix AsciiCharList NullChar
StringPrefix := 0x0D
ConstObj := ZeroOp | OneOp | OnesOp
ByteList := Nothing | <bytedata bytelist>
ByteData := 0x00 - 0xFF
WordData := ByteData[0:7] ByteData[8:15]
// 0x0000-0xFFFF
DWordData := WordData[0:15] WordData[16:31]
// 0x00000000-0xFFFFFFFF
QWordData := DWordData[0:31] DWordData[32:63]
// 0x0000000000000000-0xFFFFFFFFFFFFFFFF
AsciiCharList := Nothing | <asciichar asciicharlist>
